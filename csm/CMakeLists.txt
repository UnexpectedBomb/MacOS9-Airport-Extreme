cmake_minimum_required(VERSION 3.13)
project(APCsm C)

# The AirPort Extreme Control Strip module. Configure with the Retro68 PowerPC toolchain:
#   cmake -S . -B build \
#     -DCMAKE_TOOLCHAIN_FILE=$HOME/Retro68-build/toolchain/powerpc-apple-macos/cmake/retroppc.toolchain.cmake
#   cmake --build build
#
# The recipe is bluetooth/csm's, proven on this machine: a SHARED library exporting
# ControlStripModule (-bE, not -e, which would drop the export), MakePEF, patch the fragment's main,
# then Rez with the PEF in the DATA fork and set the bundle flag. See the os9_control_strip_module note.

set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED TRUE)
add_compile_options(-Wno-multichar -Wall)

set(CSM_VER "1.10")

# ap_status.h is the driver's own header, included in place -- ONE definition of the block for the
# driver, this module and the panel, so the three cannot drift. Its _Static_asserts pin the layout.
include_directories(${CMAKE_CURRENT_SOURCE_DIR}/../probe)

add_library(APCsm SHARED ap_csm.c)
if(TARGET retrocrt)
    add_dependencies(APCsm retrocrt)
endif()
set_target_properties(APCsm PROPERTIES
    LINK_FLAGS "-Wl,-bE:${CMAKE_CURRENT_SOURCE_DIR}/ap_csm.exp")
target_link_libraries(APCsm ControlStripLib)

add_custom_command(OUTPUT APCsm.pef
    COMMAND ${MAKE_PEF} libAPCsm.so -o APCsm.pef
    COMMAND /usr/bin/env python3 ${CMAKE_CURRENT_SOURCE_DIR}/scripts/patch-pef-main.py
            APCsm.pef ControlStripModule
    DEPENDS APCsm ${CMAKE_CURRENT_SOURCE_DIR}/scripts/patch-pef-main.py)

if(REZ_INCLUDE_PATH)
    set(REZ_INCLUDE_FLAG -I${REZ_INCLUDE_PATH})
endif()
set(SET_ICON ${CMAKE_CURRENT_SOURCE_DIR}/scripts/set-custom-icon.py)

# The Finder icon: ap_csm_icon.r (generated by ../art/make-art.py) is a second Rez INPUT rather than an
# #include, so it is a real DEPENDS -- an #included .r is invisible to CMake and a changed icon would be
# silently skipped (feedback_rez_include_deps). kHasCustomIcon (the default here) pairs with its -16455
# family; --bundle is independent and required for option-drag out of the strip.
add_custom_command(OUTPUT AirPortExtremeStrip-v${CSM_VER}.bin
    COMMAND ${REZ} ${REZ_INCLUDE_FLAG}
                   ${CMAKE_CURRENT_SOURCE_DIR}/ap_csm.r
                   ${CMAKE_CURRENT_SOURCE_DIR}/ap_csm_icon.r
                   -o AirPortExtremeStrip-v${CSM_VER}.raw.bin
                   -t sdev -c APXs
                   --data APCsm.pef
    COMMAND /usr/bin/env python3 ${SET_ICON}
            AirPortExtremeStrip-v${CSM_VER}.raw.bin AirPortExtremeStrip-v${CSM_VER}.bin
            --name "AirPort Extreme Strip" --bundle
    DEPENDS APCsm.pef ${CMAKE_CURRENT_SOURCE_DIR}/ap_csm.r ${CMAKE_CURRENT_SOURCE_DIR}/ap_csm_icon.r ${SET_ICON})
add_custom_target(APCsm_sdev ALL DEPENDS AirPortExtremeStrip-v${CSM_VER}.bin)

# ⚠ Both 'vers' strings must carry CSM_VER: Get Info shows the long one, list views the short one, and
# a half-bumped release (the two disagreeing) has shipped in this project before.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/ap_csm.r)
file(READ ${CMAKE_CURRENT_SOURCE_DIR}/ap_csm.r _rsrc)
string(FIND "${_rsrc}" "\"${CSM_VER}\"" _short)
string(FIND "${_rsrc}" "\"${CSM_VER} - " _long)
if(_short EQUAL -1 OR _long EQUAL -1)
    message(FATAL_ERROR "ap_csm.r 'vers' strings must both carry CSM_VER ${CSM_VER} (short=${_short}, long=${_long})")
endif()
