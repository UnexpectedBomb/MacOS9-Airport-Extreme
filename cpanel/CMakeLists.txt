cmake_minimum_required(VERSION 3.13)
project(APPanel C)

# The AirPort Extreme control panel. Configure with the Retro68 PowerPC toolchain:
#   cmake -S . -B build \
#     -DCMAKE_TOOLCHAIN_FILE=$HOME/Retro68-build/toolchain/powerpc-apple-macos/cmake/retroppc.toolchain.cmake
#   cmake --build build
#
# An APPLICATION in the Control Panels folder, as bluetooth/cpanel is and as Apple's own control panels are
# from Mac OS 8.5 on: the 'cdev' protocol is absent from these interfaces. Installed as "AirPort Extreme",
# creator 'APXp' -- the creator the strip module's "Open AirPort Extreme" looks for.
#
# Host test for the view logic (every string and show/hide decision), run before every build:
#   cc -std=c99 -Wall -I../probe -o view_test view_test.c && ./view_test

set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED TRUE)
add_compile_options(-Wno-multichar -Wall)

set(PANEL_VER "1.4.8")

# ap_status.h is the driver's own header, included in place: ONE definition of the block for the driver,
# the strip module and this panel. Its _Static_asserts pin the layout.
include_directories(${CMAKE_CURRENT_SOURCE_DIR}/../probe)

# ap_panel_icon.r (the user's icon, generated by ../art/make-art.py) is a SEPARATE .r source, not an
# #include: add_application Rezzes each .r with its own DEPENDS, so a changed icon re-Rezzes. An #included
# .r would be invisible to CMake and the old icon would ship silently (feedback_rez_include_deps).
add_application(AirPortExtremePanel
    ap_panel.c
    ap_panel.r
    ap_panel_icon.r
    ap_panel_note.r
    ap_panel_wifi.r
    TYPE     "APPL"
    CREATOR  "APXp"
)

# AppearanceLib: the theme calls (group frames, fonts, colours, StandardAlert) are not in InterfaceLib.
# NameRegistryLib: the "is the card in the machine" walk. Network Setup is NOT linked -- ap_panel.c loads
# CfgOpenTpt at run time, so a Mac without it loses one status line rather than the whole panel.
target_link_libraries(AirPortExtremePanel PRIVATE AppearanceLib NameRegistryLib)

if(TARGET retrocrt)
    add_dependencies(AirPortExtremePanel retrocrt)
endif()

# ★ THE DEMO BUILD: the same panel with a fake status block that walks every state (AP_PANEL_DEMO in
# ap_panel.c), for looking at the window in QEMU with no card and no driver. ⚠ NEVER STAGED: it would show
# made-up networks on a real Mac. No bundle, no rename, so it cannot be mistaken for the real file.
add_application(AirPortExtremePanelDemo
    ap_panel.c
    ap_panel.r
    ap_panel_icon.r
    ap_panel_note.r
    ap_panel_wifi.r
    TYPE     "APPL"
    CREATOR  "APXd"
)
target_compile_definitions(AirPortExtremePanelDemo PRIVATE AP_PANEL_DEMO=1)
target_link_libraries(AirPortExtremePanelDemo PRIVATE AppearanceLib NameRegistryLib)
if(TARGET retrocrt)
    add_dependencies(AirPortExtremePanelDemo retrocrt)
endif()

# ⚠ Rez has no option for Finder flags, and add_application names the MacBinary after the TARGET, which
# cannot contain a space. The strip module's script sets kHasBundle and kHasCustomIcon (the icon family is
# at -16455 and 128 in ap_panel_icon.r) and renames it "AirPort Extreme", writing a NEW file.
# ⚠ Attached to AirPortExtremePanel_APPL, not AirPortExtremePanel: add_application Rezzes the .bin in
# that separate target, and a POST_BUILD on the other runs before the .bin exists.
set(SET_ICON ${CMAKE_CURRENT_SOURCE_DIR}/../csm/scripts/set-custom-icon.py)
add_custom_command(TARGET AirPortExtremePanel_APPL POST_BUILD
    COMMAND /usr/bin/env python3 ${SET_ICON}
            ${CMAKE_CURRENT_BINARY_DIR}/AirPortExtremePanel.bin
            ${CMAKE_CURRENT_BINARY_DIR}/AirPortExtremePanel-v${PANEL_VER}.bin
            --name "AirPort Extreme" --bundle
    COMMENT "AirPortExtremePanel-v${PANEL_VER}.bin: named \"AirPort Extreme\", kHasBundle + kHasCustomIcon")

# ⚠ Every 'vers' string must carry PANEL_VER: Get Info shows the long one, list views the short one, and a
# half-bumped release has shipped in this project before.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    ${CMAKE_CURRENT_SOURCE_DIR}/ap_panel.r ${CMAKE_CURRENT_SOURCE_DIR}/ap_panel.c)
file(READ ${CMAKE_CURRENT_SOURCE_DIR}/ap_panel.r _rsrc)
string(REGEX MATCHALL "\"${PANEL_VER}\"" _short "${_rsrc}")
string(REGEX MATCHALL "\"${PANEL_VER} - " _long "${_rsrc}")
list(LENGTH _short _nshort)
list(LENGTH _long _nlong)
if(NOT _nshort EQUAL 2 OR NOT _nlong EQUAL 2)
    message(FATAL_ERROR "ap_panel.r: both 'vers' resources must carry PANEL_VER ${PANEL_VER} in both strings "
                        "(short=${_nshort}/2, long=${_nlong}/2)")
endif()
# ...and the About box, which says the version in ap_panel.c.
file(READ ${CMAKE_CURRENT_SOURCE_DIR}/ap_panel.c _src)
string(FIND "${_src}" "#define kPanelVersion  \"${PANEL_VER}\"" _about)
if(_about EQUAL -1)
    message(FATAL_ERROR "ap_panel.c: kPanelVersion must be \"${PANEL_VER}\" (the About box shows it)")
endif()
